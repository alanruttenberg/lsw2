(defun manchester-logical-axioms-for-entity-text (class ont kind &optional (include-referencing nil))
  (setq ont (or (v3kb-weakened-from ont) ont))
  (if (stringp class) (setq class (make-uri class)))
  (if (or (not (get-entity class kind ont))) (equal class !owl:Nothing))
      nil
      (remove-duplicates
       (let* ((direct-axioms (get-rendered-referencing-axioms class kind ont t))
	      (referenced-axioms (and include-referencing (set-difference (get-rendered-referencing-axioms class kind ont) direct-axioms :key 'second))))
	 (loop for (sentence ax) in (append direct-axioms referenced-axioms)
	       unless (or (simple-subclassof-axiom? ax)
                          (simple-supropertyof-axiom? ax)
		          (jinstance-of-p ax (find-java-class 'OWLDeclarationAxiom )))
	    collect sentence)
         )
       :test 'equal))
